1. Még nem vizsgált állóvizek nézete
CREATE OR REPLACE VIEW v_meretlen_allovizek AS
SELECT a.id,
       a.nev,
       a.tipus     
FROM alloviz a
  WHERE NOT EXISTS ( SELECT * FROM vizminoseg v WHERE v.allovizid = a.id );

2. Aktuális (2026-os) mérések nézete
CREATE OR REPLACE VIEW v_2026_meresek AS
SELECT a.id,
       a.nev,
       a.tipus,
       v.MERES_DATUM
FROM alloviz a
LEFT JOIN vizminoseg v
  ON a.id = v.allovizid
  WHERE v.MERES_DATUM LIKE '2026%'
GROUP BY a.id, a.nev, a.tipus, v.MERES_DATUM;

3. Nagyobb állóvizek és vízminőségük nézete
CREATE OR REPLACE VIEW v_nagy_tavak_minosege AS
SELECT 
       a.nev,
       a.TERULET ,  
       V.MERES_DATUM,
       V.PH_ERTEK
FROM alloviz a LEFT JOIN VIZMINOSEG V on a.id = V.ALLOVIZID 
where a.terulet > 10
  

4. Vizsgálat alatt álló mérések nézete
CREATE OR REPLACE VIEW  v_vizsgalat_alatt AS
SELECT 
       a.NEV,
       v.*
FROM ALLOVIZ a INNER JOIN VIZMINOSEG V on a.id = V.ALLOVIZID 
WHERE v.ALLAPOT ='vizsgalat_alatt'
  

5. Típusonkénti mérésszám nézete
CREATE OR REPLACE VIEW v_tipus_meresszam AS
SELECT NVL(a.tipus, 'Ismeretlen') AS tipus,
       COUNT(v.id) AS meres_db
FROM alloviz a
LEFT JOIN vizminoseg v 
  ON a.id = v.allovizid
GROUP BY a.tipus;
